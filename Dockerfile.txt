# Secure, Hardened Dockerfile for AI Software Factory Sandbox Execution Environment
FROM python:3.12-slim

# Prevent interactive prompts
ENV DEBIAN_FRONTEND=noninteractive

# Install core build dependencies, git, and secure utils
RUN apt-get update && apt-get install -y --no-install-recommends     git     build-essential     && rm -rf /var/lib/apt/lists/*

# Install python analysis and testing tools
RUN pip install --no-cache-dir     flake8     pytest     tree-sitter

# Add non-root security user
RUN useradd -m -s /bin/bash sandbox

# Configure execution working directory
WORKDIR /workspace
RUN chown -R sandbox:sandbox /workspace

# Switch to non-root sandbox execution user
USER sandbox

# Default execution entrypoint
CMD ["/bin/bash"]
